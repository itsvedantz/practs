import pandas as pd


customers = pd.DataFrame({
    'customer_id': [1, 2, 2, 5],
    'customer_name': ['Alice', 'Bob', 'Charlie', 'David']
})


orders = pd.DataFrame({
    'customer_id': [1, 2, 3, 4],
    'order_amount': [250, 450, 300, 500]
})


print("Customers DataFrame:")
print(customers)
print("\nOrders DataFrame:")
print(orders)


inner_join = pd.merge(customers, orders, on="customer_id", how="inner")
print("\n--- INNER JOIN ---")
print(inner_join)


left_join = pd.merge(customers, orders, on="customer_id", how="left")
print("\n--- LEFT JOIN ---")
print(left_join)


right_join = pd.merge(customers, orders, on="customer_id", how="right")
print("\n--- RIGHT JOIN ---")
print(right_join)


outer_join = pd.merge(customers, orders, on="customer_id", how="outer")
print("\n--- OUTER JOIN ---")
print(outer_join)
