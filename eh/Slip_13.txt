import math
from collections import Counter

documents = [
"Document about python programming language and data analysis",
"Document discussing machine learning algorithms and programming techniques",
"Overview of natural language processing and its applications"
]

query = "python programming"

def preprocess(text):
    return text.lower().split()

docs_tokens = [preprocess(doc) for doc in documents]
query_tokens = preprocess(query)

vocab = set()
for doc in docs_tokens:
    vocab.update(doc)
vocab = list(vocab)

def compute_tf(tokens):
    tf = {}
    count = Counter(tokens)
    for word in vocab:
        tf[word] = count[word] / len(tokens)
    return tf

tf_docs = [compute_tf(doc) for doc in docs_tokens]

def compute_idf():
    idf = {}
    N = len(docs_tokens)
    for word in vocab:
        df = sum(1 for doc in docs_tokens if word in doc)
        idf[word] = math.log(N / (df + 1))
    return idf

idf = compute_idf()

tfidf_docs = []
for tf in tf_docs:
    tfidf = {}
    for word in vocab:
        tfidf[word] = tf[word] * idf[word]
    tfidf_docs.append(tfidf)

query_tf = compute_tf(query_tokens)

query_tfidf = {}
for word in vocab:
    query_tfidf[word] = query_tf[word] * idf[word]

def cosine_similarity(vec1, vec2):
    dot = sum(vec1[word] * vec2[word] for word in vocab)

    mag1 = math.sqrt(sum(vec1[word]**2 for word in vocab))
    mag2 = math.sqrt(sum(vec2[word]**2 for word in vocab))

    if mag1 == 0 or mag2 == 0:
        return 0

    return dot / (mag1 * mag2)

A. Implement Vector Space Model with TF-IDF Weighting
Given Corpus
Document 1
"Document about python programming language and data analysis."

Document 2
"Document discussing machine learning algorithms and programming techniques."

Document 3
"Overview of natural language processing and its applications."

Query
"python programming"

Steps
Tokenize the documents

Create vocabulary

Calculate TF (Term Frequency)

Calculate IDF (Inverse Document Frequency)

Compute TF-IDF weights

Represent documents as vectors

Python Program
import math
from collections import Counter

documents = [
"Document about python programming language and data analysis",
"Document discussing machine learning algorithms and programming techniques",
"Overview of natural language processing and its applications"
]

query = "python programming"

docs = [doc.lower().split() for doc in documents]
query_words = query.lower().split()

vocab = set()

for doc in docs:
    vocab.update(doc)

vocab = list(vocab)

def compute_tf(words):
    tf = {}
    count = Counter(words)

    for word in vocab:
        tf[word] = count[word] / len(words)

    return tf

tf_docs = [compute_tf(doc) for doc in docs]

N = len(docs)

idf = {}

for word in vocab:
    df = sum(1 for doc in docs if word in doc)
    idf[word] = math.log(N / (df + 1))

tfidf_docs = []

for tf in tf_docs:
    tfidf = {}
    for word in vocab:
        tfidf[word] = tf[word] * idf[word]

    tfidf_docs.append(tfidf)

print("TF-IDF vectors created for all documents.")
Explanation
TF measures how frequently a term appears in a document.

IDF reduces the weight of common words.

TF-IDF helps identify important words in a document collection.

Thus documents and queries are represented as vectors in Vector Space Model.

B. Implement Cosine Similarity
import math
from collections import Counter

query = "gold silver truck"
document = "shipment of gold damaged in a gold fire"

q_words = query.lower().split()
d_words = document.lower().split()

vocab = list(set(q_words + d_words))

q_count = Counter(q_words)
d_count = Counter(d_words)

q_vector = []
d_vector = []

for word in vocab:
    q_vector.append(q_count[word])
    d_vector.append(d_count[word])

dot_product = sum(q*d for q,d in zip(q_vector,d_vector))
