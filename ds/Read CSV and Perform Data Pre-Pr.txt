---Read CSV and Perform Data Pre-Processing students---
data <- read.csv("students.csv")

print(data)

colSums(is.na(data))

data$Age[is.na(data$Age)] <- mean(data$Age, na.rm=TRUE)
data$Marks[is.na(data$Marks)] <- mean(data$Marks, na.rm=TRUE)

print(data)

boxplot(data$Marks, main="Marks Outliers")

Q1 <- quantile(data$Marks, 0.25)
Q3 <- quantile(data$Marks, 0.75)
IQR <- Q3 - Q1

lower <- Q1 - 1.5 * IQR
upper <- Q3 + 1.5 * IQR

data <- data[data$Marks >= lower & data$Marks <= upper, ]

print(data)
