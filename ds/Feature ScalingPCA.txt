---Feature Scaling (Standardization & Normalization in R)---

data <- data.frame(
  Country = c("France","Spain","Germany","Spain","Germany","France","Spain","France","Germany","France"),
  Age = c(44,27,30,38,40,35,31,48,50,37),
  Salary = c(72000,48000,54000,61000,85000,58000,52000,79000,83000,67000),
  Purchased = c("No","Yes","No","No","Yes","Yes","No","Yes","No","Yes")
)

num_data <- data[,c("Age","Salary")]

standardized_data <- scale(num_data)

print("Standardized Data")
print(standardized_data)

normalized_data <- (num_data - min(num_data)) / (max(num_data) - min(num_data))

print("Normalized Data")
print(normalized_data)

---PCA on Iris Dataset (R)---

data(iris)

x <- iris[,1:4]

pca <- prcomp(x, scale = TRUE)

summary(pca)

pca_data <- data.frame(pca$x, Species = iris$Species)

plot(pca_data$PC1, pca_data$PC2,
     col = as.numeric(pca_data$Species),
     pch = 19,
     xlab = "Principal Component 1",
     ylab = "Principal Component 2",
     main = "PCA on Iris Dataset")
