import matplotlib.pyplot as plt
import seaborn as sns
import plotly.express as px
import pandas as pd
from sklearn.cluster import KMeans

df = pd.DataFrame({
    'gender': ['Male', 'Female', 'Female', 'Male', 'Male'],
    'churn': ['Yes', 'No', 'Yes', 'No', 'Yes'],
    'age': [25, 42, 32, 30, 35],
    'service_type': ['Permanent Staffing', 'Contract Staffing', 'Part-Time Staffing',
                     'Freelance', 'Temporary'],
    'tenure': [2, 6, 10, 5, 7]
})

# 1. Churn rate by gender
sns.countplot(data=df, x='gender', hue='churn')
plt.title("Churn rate by gender")
plt.show()

# 2. Churn rate by age group (Plotly)
fig = px.histogram(
    df,
    x='age',
    color='churn',
    nbins=20,
    histnorm='percent'
)

fig.update_layout(
    title='Churn rate by age group',
    xaxis_title='Age',
    yaxis_title='Percentage of customers'
)

fig.show()

# 3. Correlation matrix
# Convert churn to numeric if needed
df['churn_numeric'] = df['churn'].map({'Yes': 1, 'No': 0})

correlation_matrix = df.select_dtypes(include='number').corr()

sns.heatmap(correlation_matrix, annot=True, cmap='coolwarm')
plt.title('Correlation matrix')
plt.show()

# 4. Customer tenure vs churn (Plotly)
fig = px.scatter(
    df,
    x='tenure',
    y='churn_numeric',
    color='churn'
)

fig.update_layout(
    title='Customer Tenure vs Churn',
    xaxis_title='Tenure',
    yaxis_title='Churn (1 = Yes, 0 = No)'
)

fig.show()

# 5. Select numerical features
X = df[['age', 'tenure']]

# Apply KMeans
kmeans = KMeans(n_clusters=2, random_state=42, n_init=10)
df['cluster'] = kmeans.fit_predict(X)

fig = px.scatter(
    df,
    x='age',
    y='tenure',
    color='cluster',
    title='Customer Segmentation using KMeans'
)

print(df.groupby('cluster')[['age', 'tenure']].mean())

fig.show()
