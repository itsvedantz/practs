---K-Means Clustering + Elbow Method – Iris---

data <- iris[,1:4]

wss <- numeric()

for(i in 1:10){
  k <- kmeans(data,i,nstart=10)
  wss[i] <- k$tot.withinss
}

plot(1:10,wss,type="b")

set.seed(123)
kmodel <- kmeans(data,3,nstart=10)

table(kmodel$cluster)

plot(data,col=kmodel$cluster,pch=19)

---K-Means + Silhouette Method---

library(cluster)

data <- iris[,1:4]

km <- kmeans(data,3,nstart=10)

sil <- silhouette(km$cluster,dist(data))

plot(sil)
