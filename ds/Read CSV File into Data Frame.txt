---Read CSV File into Data Frame---
car_data <- read.csv("cars.csv")

print(car_data)

high_price <- car_data[car_data$SellPrice > 4000, ]

print(high_price)

sorted_data <- car_data[order(car_data$SellPrice), ]

print(sorted_data)

group_data <- aggregate(SellPrice ~ Make, data = car_data, FUN = mean)

print(group_data)

---PCA on Iris Dataset---

data(iris)

x <- iris[,1:4]

pca <- prcomp(x, scale = TRUE)

summary(pca)

pca_data <- data.frame(pca$x, Species = iris$Species)

plot(pca_data$PC1, pca_data$PC2,
     col = as.numeric(pca_data$Species),
     pch = 19,
     xlab = "PC1",
     ylab = "PC2",
     main = "PCA on Iris Dataset")
