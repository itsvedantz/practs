---Logistic Regression – Iris (Binary)---

data(iris)

iris$Species <- ifelse(iris$Species=="setosa",1,0)

model <- glm(Species~Sepal.Length+Sepal.Width+Petal.Length+Petal.Width,family=binomial,data=iris)

pred <- predict(model,type="response")

class <- ifelse(pred>0.5,1,0)

table(class,iris$Species)
