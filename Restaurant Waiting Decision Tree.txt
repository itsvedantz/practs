---Restaurant Waiting Decision Tree---
data <- data.frame(
  Alt=c("Yes","Yes","No","Yes","Yes","No","No","No","No"),
  Bar=c("No","No","Yes","No","No","Yes","Yes","No","Yes"),
  Fri=c("No","No","No","Yes","Yes","No","No","No","Yes"),
  Hun=c("Yes","Yes","No","Yes","No","Yes","No","Yes","No"),
  Pat=c("Some","Full","Some","Full","Full","Some","None","Some","Full"),
  Price=c(1200,2500,2200,1245,4300,3400,1000,3200,3400),
  Rain=c("No","No","No","No","No","Yes","Yes","Yes","Yes"),
  Res=c("Yes","No","No","No","Yes","Yes","No","Yes","No"),
  Type=c("French","Thai","Burger","Thai","French","Italian","Burger","Thai","Burger"),
  Est=c("0-10","30-60","0-10","30-60",">60","0-10","0-10","0-10",">60"),
  Wait=c("Yes","No","Yes","Yes","No","Yes","No","Yes","No")
)

print(data)

install.packages("rpart")
library(rpart)
library(rpart.plot)

model <- rpart(Wait ~ Alt + Bar + Fri + Hun + Pat + Price + Rain + Res + Type + Est,
               data=data,
               method="class")

print(model)

rpart.plot(model)

pred <- predict(model, data, type="class")

print(pred)

table(Predicted=pred, Actual=data$Wait)
