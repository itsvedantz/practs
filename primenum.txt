@WebMethod(operationName = "checkPrime")
public String checkPrime(@WebParam(name = "n") int n) {
    if (n <= 1) {
        return "Number is not Prime";
    }
    
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            return "Number is not Prime";
        }
    }
    
    return "Number is Prime";
}
