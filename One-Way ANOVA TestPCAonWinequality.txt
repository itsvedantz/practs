---One-Way ANOVA Test---
classA <- c(85, 90, 88, 82, 87)
classB <- c(76, 78, 80, 81, 75)
classC <- c(92, 88, 94, 89, 90)

scores <- c(classA, classB, classC)

group <- factor(c(rep("A",5), rep("B",5), rep("C",5)))

data <- data.frame(scores, group)

anova_result <- aov(scores ~ group, data=data)

summary(anova_result)
